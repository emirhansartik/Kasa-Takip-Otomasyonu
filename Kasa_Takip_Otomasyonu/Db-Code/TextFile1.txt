-- 1. Kullanıcılar Tablosu (Giriş Ekranı İçin)
CREATE TABLE Kullanicilar (
    KullaniciId INT IDENTITY(1,1) PRIMARY KEY,
    AdSoyad NVARCHAR(100) NOT NULL,
    KullaniciAdi NVARCHAR(50) NOT NULL UNIQUE,
    Sifre NVARCHAR(50) NOT NULL
);

-- 2. Kategoriler Tablosu
CREATE TABLE Kategoriler (
    KategoriId INT IDENTITY(1,1) PRIMARY KEY,
    KategoriAd NVARCHAR(100) NOT NULL
);

-- 3. Kasalar Tablosu (Nakit, Banka vb.)
CREATE TABLE Kasalar (
    KasaId INT IDENTITY(1,1) PRIMARY KEY,
    KasaAd NVARCHAR(100) NOT NULL,
    KasaTuru NVARCHAR(50) NOT NULL, -- Nakit, Banka, Kredi Kartı
    Bakiye DECIMAL(18,2) NOT NULL DEFAULT 0
);

-- 4. Gelir ve Giderler Tablosu (Birleştirilmiş Temiz Yapı)
CREATE TABLE GelirGiderler (
    IslemId INT IDENTITY(1,1) PRIMARY KEY,
    Baslik NVARCHAR(150) NOT NULL,
    Tutar DECIMAL(18,2) NOT NULL,
    IslemTuru NVARCHAR(20) NOT NULL, -- 'Gelir' veya 'Gider'
    Tarih DATETIME NOT NULL DEFAULT GETDATE(),
    KategoriId INT FOREIGN KEY REFERENCES Kategoriler(KategoriId),
    KasaId INT FOREIGN KEY REFERENCES Kasalar(KasaId),
    Aciklama NVARCHAR(250) NULL
);

-- 5. Kasa Hareketleri (Transfer ve Hesap Hareketleri)
CREATE TABLE KasaHareketleri (
    HareketId INT IDENTITY(1,1) PRIMARY KEY,
    KasaId INT FOREIGN KEY REFERENCES Kasalar(KasaId),
    HareketTuru NVARCHAR(50) NOT NULL, -- 'Gelen Havale', 'Giden EFT', 'Nakit Giriş' vb.
    Tutar DECIMAL(18,2) NOT NULL,
    Tarih DATETIME NOT NULL DEFAULT GETDATE(),
    Aciklama NVARCHAR(250) NULL
);

-- 6. Bütçe Limitleri Tablosu
CREATE TABLE ButceLimitleri (
    LimitId INT IDENTITY(1,1) PRIMARY KEY,
    KategoriId INT FOREIGN KEY REFERENCES Kategoriler(KategoriId),
    AylikLimit DECIMAL(18,2) NOT NULL,
    Donem NVARCHAR(20) NOT NULL -- Örn: '2026-11'
);
GO
